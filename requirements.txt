# ---- Agent framework ----
# Pins match the versions verified in the local venv. crewai 1.x is proven
# against the current litellm/mcp/fastmcp stack; the earlier 0.86.0 pins were
# left over from a previous environment and untested against the newer clients.
crewai==1.15.7
crewai-tools==1.15.7
litellm==1.67.5

# ---- LLM framework ----
# langchain 1.x is required for openai 2.x compatibility. crewai 1.15.7 pins
# openai>=2.30.0, but the langchain 0.3.x stack (langchain-openai<=0.3) pins
# openai<2.0.0 — a hard conflict that made `pip install` unresolvable. The 1.x
# line tracks openai 2.x and is proven against crewai 1.15.7 / litellm 1.67.5.
langchain==1.3.14
langchain-community==0.4.2
langchain-openai==1.4.1
langchain-text-splitters==1.1.2

# ---- Backend ----
# fastapi must be >=0.135 (starlette>=0.46, no upper bound) to coexist with
# fastmcp 3.4.4, which requires starlette>=1.0.1. fastapi 0.115.x pins
# starlette<0.42, a hard conflict in the same venv.
fastapi==0.135.0
uvicorn[standard]==0.35.0
pydantic==2.12.5
pydantic-settings==2.14.2
# fastmcp-slim 3.4.4 (server extra) requires python-multipart>=0.0.26.
python-multipart==0.0.26
# EmailStr validation in request schemas.
email-validator==2.2.0

# ---- Auth ----
python-jose[cryptography]==3.3.0
passlib[bcrypt]==1.7.4
# passlib 1.7.4 reads bcrypt.__about__, which bcrypt 4.1 removed.
bcrypt==4.0.1

# ---- Database ----
sqlalchemy==2.0.36

# ---- Vector database ----
chromadb==1.1.1

# ---- Embeddings (EMBEDDING_PROVIDER=local) ----
# ONNX on CPU. Deliberately not sentence-transformers, which pulls ~2GB of torch.
fastembed==0.5.1

# ---- MCP ----
# mcp is the official SDK; fastmcp is the ergonomic server/client layer used by
# mcp_server/server.py and mcp_server/test_client.py. Both are imported
# directly, so both are pinned explicitly.
mcp==1.28.1
fastmcp==3.4.4

# ---- Document ingestion (PDF / DOCX / TXT) ----
pypdf==5.1.0
docx2txt==0.8
# Chunk length is measured in tokens, not characters (see docs/RAG.md).
tiktoken==0.12.0

# ---- Config ----
python-dotenv==1.2.2

# ---- Runtime prerequisites ----
# setuptools is a runtime requirement for crewai (it imports pkg_resources at
# import time in telemetry). Modern venvs no longer seed it by default, and
# version 82+ dropped pkg_resources entirely, so the upper bound is load-bearing.
setuptools>=70.0.0,<82.0.0

# ---- Transitive pins ----
# httpx is pinned to the version crewai/litellm/chromadb resolve to in the
# tested venv. Keep in sync if you bump any of those.
httpx==0.28.1
